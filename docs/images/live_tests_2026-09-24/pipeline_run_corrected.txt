"""Calculate the GC content of a DNA sequence."""

import argparse
from collections.abc import Sequence


def gc_content(sequence: str) -> float:
    """Return the percentage of G and C bases in a DNA sequence."""
    normalized = "".join(sequence.split()).upper()

    if not normalized:
        raise ValueError("DNA sequence must not be empty")

    invalid_bases = set(normalized) - set("ACGT")
    if invalid_bases:
        raise ValueError(
            f"invalid DNA base(s): {', '.join(sorted(invalid_bases))}"
        )

    gc_bases = normalized.count("G") + normalized.count("C")
    return gc_bases / len(normalized) * 100


def main(argv: Sequence[str] | None = None) -> int:
    """Run the command-line interface."""
    parser = argparse.ArgumentParser(
        description="Calculate the GC-content percentage of a DNA sequence."
    )
    parser.add_argument("sequence", help="DNA sequence containing A, C, G, and T")
    args = parser.parse_args(argv)

    try:
        percentage = gc_content(args.sequence)
    except ValueError as error:
        parser.error(str(error))

    print(f"{percentage:.2f}%")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Tests for gc_content.py."""

import contextlib
import io
import unittest

from gc_content import gc_content, main


class GCContentTests(unittest.TestCase):
    def test_all_gc_bases(self):
        self.assertEqual(gc_content("GCGC"), 100.0)

    def test_no_gc_bases(self):
        self.assertEqual(gc_content("AATT"), 0.0)

    def test_mixed_bases(self):
        self.assertEqual(gc_content("ACGT"), 50.0)

    def test_is_case_insensitive(self):
        self.assertEqual(gc_content("acgT"), 50.0)

    def test_ignores_whitespace(self):
        self.assertEqual(gc_content("AC GT\n"), 50.0)

    def test_empty_sequence_raises_value_error(self):
        with self.assertRaisesRegex(ValueError, "must not be empty"):
            gc_content(" \n\t")

    def test_invalid_base_raises_value_error(self):
        with self.assertRaisesRegex(ValueError, r"invalid DNA base\(s\): N"):
            gc_content("ACNT")

    def test_cli_prints_percentage(self):
        output = io.StringIO()
        with contextlib.redirect_stdout(output):
            result = main(["ACGT"])

        self.assertEqual(result, 0)
        self.assertEqual(output.getvalue(), "50.00%\n")


if __name__ == "__main__":
    unittest.main()
